**Goal:** Extract lore's engine into a shared `@lore/core` package, then scaffold **Atlas** — a local Next.js app that browses lore-typed markdown docs across multiple project repos (read-only).

**Tech Stack:** TypeScript/ESM, pnpm, Vitest, Next.js 15 (App Router) + React 19 + Tailwind, `marked` for markdown rendering. Spec: `/Users/suki/dev/cliphy/docs/superpowers/specs/2026-06-09-workspace-app-design.md`.

- [ ] **Step 5: Implement `src/lib/projects.ts`.**

```ts
import { readFileSync } from "node:fs";
import { join } from "node:path";
import { resolveConfig, readDocs, type Doc, type DocTypeSchema, type LoreConfig } from "@lore/core";

export interface Project {
  name: string;
  path: string; // repo root
  docsRoot: string;
  types: Record<string, DocTypeSchema>;
}

interface AtlasConfig {
  projects: { name: string; path: string }[];
}

function configPath(): string {
  return join(process.cwd(), "atlas.config.json");
}

export function loadProjects(): Project[] {
  const cfg = JSON.parse(readFileSync(configPath(), "utf8")) as AtlasConfig;
  return cfg.projects.map((p) => {
    const lore: LoreConfig = resolveConfig(p.path);
    return { name: p.name, path: p.path, docsRoot: lore.docsRoot, types: lore.types };
  });
}

export function getProject(name: string): Project {
  const project = loadProjects().find((p) => p.name === name);
  if (!project) throw new Error(`unknown project: ${name}`);
  return project;
}

export function getDocs(projectName: string, typeName: string): Doc[] {
  const project = getProject(projectName);
  const schema = project.types[typeName];
  if (!schema) throw new Error(`unknown type: ${typeName}`);
  return readDocs(join(project.docsRoot, schema.dir), schema);
}

export function getDoc(projectName: string, typeName: string, fileBase: string): Doc | undefined {
  return getDocs(projectName, typeName).find((d) => d.file === `${fileBase}.md`);
}
```

```ts
export function cellText(doc: Doc, source: string, format?: string): string {
  const raw =
    source === "id" ? (Number.isNaN(doc.id) ? "" : String(doc.id)) : doc.frontmatter[source];
  if (format === "date") {
    if (raw instanceof Date) return raw.toISOString().slice(0, 10);
    return raw == null ? "—" : String(raw);
  }
  if (format === "yesno") return raw ? "yes" : "no";
  return raw == null || raw === "" ? "—" : String(raw);
}
```

```tsx
import { marked } from "marked";
import { getProject, getDoc, cellText } from "../../../../../lib/projects";

export const dynamic = "force-dynamic";

export default async function DocPage({
  params,
}: {
  params: Promise<{ project: string; type: string; file: string }>;
}) {
  const { project, type, file } = await params;
  const schema = getProject(project).types[type];
  const doc = getDoc(project, type, file);
  if (!doc) return <div className="text-zinc-400">Not found.</div>;
  const html = marked.parse(doc.body) as string;
  return (
    <article className="space-y-6">
      <header className="space-y-3">
        <h1 className="text-2xl font-semibold text-zinc-100">
          {String(doc.frontmatter.title ?? doc.slug)}
        </h1>
        <dl className="flex flex-wrap gap-x-6 gap-y-1 text-sm text-zinc-400">
          {schema.frontmatter
            .filter((f) => f.name !== "title")
            .map((f) => (
              <div key={f.name} className="flex gap-1.5">
                <dt className="text-zinc-500">{f.name}:</dt>
                <dd>{cellText(doc, f.name, f.type === "date" ? "date" : undefined)}</dd>
              </div>
            ))}
        </dl>
      </header>
      <div
        className="prose prose-invert prose-zinc max-w-none prose-headings:font-semibold"
        dangerouslySetInnerHTML={{ __html: html }}
      />
    </article>
  );
}
```
