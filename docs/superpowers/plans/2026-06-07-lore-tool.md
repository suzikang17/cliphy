> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build `lore` v1 — an in-repo TUI + CLI over structured markdown, handling one doc type (decisions), dogfooded against Cliphy's existing 35 decisions.

**Architecture:** One pure-functions **core** (parse/validate/id/index/search) with thin **CLI** and **TUI** frontends. Markdown files are the single source of truth; the doc-type schema lives in the target repo; every frontend is a stateless regenerable view + thin writes through the core.

**Tech Stack:** TypeScript (ESM, Node 22), pnpm, Vitest, `gray-matter` (frontmatter), `yaml` (schema files), `zod` (internal validation), `ink` + `react` (TUI). Built with `tsc`, dev-run with `tsx`.

**Where it's built:** New standalone repo at `~/dev/lore`. All paths below are relative to that repo root unless absolute. Dogfood target is the Cliphy docs at `/Users/suki/dev/cliphy/docs`.

```ts
import matter from "gray-matter";
import type { Doc } from "./types.js";

const ID_RE = /^(\d+)-(.+)\.md$/;

export function parseDoc(file: string, raw: string): Doc {
  const { data, content } = matter(raw);
  const m = file.match(ID_RE);
  const id = m ? parseInt(m[1], 10) : NaN;
  const slug = m ? m[2] : file.replace(/\.md$/, "");
  return { id, slug, file, frontmatter: data, sections: extractSections(content), raw };
}

function extractSections(content: string): Record<string, string> {
  const sections: Record<string, string> = {};
  const lines = content.split("\n");
  let current: string | null = null;
  let buf: string[] = [];
  const flush = () => {
    if (current) sections[current] = buf.join("\n").trim();
  };
  for (const line of lines) {
    const h2 = line.match(/^##\s+(.+?)\s*$/);
    if (h2) {
      flush();
      current = h2[1];
      buf = [];
    } else if (current) buf.push(line);
  }
  flush();
  return sections;
}

export function serializeDoc(
  frontmatter: Record<string, unknown>,
  title: string,
  sections: Record<string, string>,
  order: string[],
): string {
  const body = [`# ${title}`, ""];
  for (const name of order) {
    const text = sections[name];
    if (text && text.trim()) body.push(`## ${name}`, "", text.trim(), "");
  }
  return matter.stringify(body.join("\n") + "\n", frontmatter);
}
```

```ts
export function slugify(s: string): string {
  return s
    .toLowerCase()
    .replace(/[^\w\s-]/g, "")
    .replace(/[\s_]+/g, "-")
    .replace(/-+/g, "-")
    .replace(/^-|-$/g, "")
    .slice(0, 60);
}
export function nextId(ids: number[]): number {
  return ids.length ? Math.max(...ids) + 1 : 1;
}
export function padId(n: number): string {
  return String(n).padStart(4, "0");
}
```

```ts
import { parse as parseYaml } from "yaml";
import { docTypeSchema, type DocTypeSchema, type Doc, type ValidationIssue } from "./types.js";

export function loadSchema(yamlText: string): DocTypeSchema {
  return docTypeSchema.parse(parseYaml(yamlText));
}

export function validateDoc(doc: Doc, schema: DocTypeSchema): ValidationIssue[] {
  const issues: ValidationIssue[] = [];
  for (const field of schema.frontmatter) {
    const val = doc.frontmatter[field.name];
    if (field.required && (val === undefined || val === null || val === "")) {
      issues.push({ file: doc.file, message: `missing required field "${field.name}"` });
      continue;
    }
    if (
      val !== undefined &&
      field.type === "enum" &&
      field.values &&
      !field.values.includes(String(val))
    ) {
      issues.push({
        file: doc.file,
        message: `field "${field.name}" must be one of ${field.values.join(", ")}`,
      });
    }
  }
  return issues;
}
```

```ts
import { readdirSync, readFileSync, writeFileSync } from "node:fs";
import { join } from "node:path";
import { parseDoc, serializeDoc } from "./frontmatter.js";
import { nextId, padId, slugify } from "./ids.js";
import type { Doc } from "./types.js";

export function readDocs(dir: string): Doc[] {
  return readdirSync(dir)
    .filter((f) => f.endsWith(".md") && f.toLowerCase() !== "index.md")
    .map((f) => parseDoc(f, readFileSync(join(dir, f), "utf8")))
    .sort((a, b) => a.id - b.id);
}

export interface NewDoc {
  frontmatter: Record<string, unknown>;
  title: string;
  sections: Record<string, string>;
  order: string[];
}

export function writeDoc(dir: string, doc: NewDoc): string {
  const ids = readDocs(dir).map((d) => d.id);
  const id = nextId(ids);
  const file = `${padId(id)}-${slugify(doc.title)}.md`;
  writeFileSync(join(dir, file), serializeDoc(doc.frontmatter, doc.title, doc.sections, doc.order));
  return file;
}
```

```ts
import { padId } from "./ids.js";
import type { Doc } from "./types.js";

export function renderIndex(docs: Doc[], heading: string): string {
  const lines = [
    `# ${heading}`,
    "",
    "Generated by lore — do not edit by hand.",
    "",
    "| # | Date | Decision | Category | Revisit? |",
    "|---|---|---|---|---|",
  ];
  for (const d of [...docs].sort((a, b) => a.id - b.id)) {
    const fm = d.frontmatter as Record<string, unknown>;
    const date = fm.date ?? "—";
    const title = fm.title ?? d.slug;
    const cat = fm.category ?? "—";
    const revisit = fm.revisit ? "yes" : "no";
    lines.push(`| ${padId(d.id)} | ${date} | [${title}](${d.file}) | ${cat} | ${revisit} |`);
  }
  return lines.join("\n") + "\n";
}
```

```ts
import type { Doc } from "./types.js";

export function searchDocs(docs: Doc[], query: string): Doc[] {
  const q = query.trim().toLowerCase();
  if (!q) return docs;
  return docs.filter((d) => {
    const hay = [String(d.frontmatter.title ?? ""), d.slug, ...Object.values(d.sections)]
      .join("\n")
      .toLowerCase();
    return hay.includes(q);
  });
}
```

```ts
import { existsSync, readdirSync, readFileSync } from "node:fs";
import { dirname, join } from "node:path";
import { loadSchema } from "./core/schema.js";
import type { DocTypeSchema } from "./core/types.js";

export interface LoreConfig {
  docsRoot: string;
  types: Record<string, DocTypeSchema>;
}

export function resolveConfig(start: string): LoreConfig {
  let dir = start;
  while (!existsSync(join(dir, ".lore", "types"))) {
    const parent = dirname(dir);
    if (parent === dir)
      throw new Error("no docs/.lore/types found (run inside a lore-enabled repo)");
    dir = parent;
  }
  const typesDir = join(dir, ".lore", "types");
  const types: Record<string, DocTypeSchema> = {};
  for (const f of readdirSync(typesDir).filter((f) => f.endsWith(".yaml"))) {
    const schema = loadSchema(readFileSync(join(typesDir, f), "utf8"));
    types[schema.name] = schema;
  }
  return { docsRoot: dir, types };
}
```

```ts
import { writeFileSync, readFileSync } from "node:fs";
import { join } from "node:path";
import { resolveConfig } from "../config.js";
import { readDocs, writeDoc } from "../core/store.js";
import { renderIndex } from "../core/index-md.js";
import { validateDoc } from "../core/schema.js";
import { searchDocs } from "../core/search.js";
import type { DocTypeSchema } from "../core/types.js";

function parseFlags(args: string[]) {
  const fields: Record<string, string> = {};
  const sections: Record<string, string> = {};
  let title = "";
  for (let i = 0; i < args.length; i++) {
    if (args[i] === "--title") title = args[++i];
    else if (args[i] === "--field") {
      const [k, ...v] = args[++i].split("=");
      fields[k] = v.join("=");
    } else if (args[i] === "--section") {
      const [k, ...v] = args[++i].split("=");
      sections[k] = v.join("=");
    }
  }
  return { title, fields, sections };
}

function dirFor(docsRoot: string, schema: DocTypeSchema) {
  return join(docsRoot, schema.dir);
}
function reindex(docsRoot: string, schema: DocTypeSchema) {
  const dir = dirFor(docsRoot, schema);
  const docs = readDocs(dir);
  writeFileSync(
    join(dir, "INDEX.md"),
    renderIndex(docs, `${schema.name[0].toUpperCase()}${schema.name.slice(1)}s`),
  );
}

const [cmd, typeName, ...rest] = process.argv.slice(2);
const cfg = resolveConfig(process.cwd());
const schema = cfg.types[typeName];
if (cmd !== "list" || !schema) {
  /* fallthrough handled below */
}
if (!schema) {
  console.error(`unknown type: ${typeName}`);
  process.exit(1);
}
const dir = dirFor(cfg.docsRoot, schema);

switch (cmd) {
  case "list": {
    for (const d of readDocs(dir)) console.log(`${d.file}  ${d.frontmatter.title ?? d.slug}`);
    break;
  }
  case "search": {
    for (const d of searchDocs(readDocs(dir), rest.join(" ")))
      console.log(`${d.file}  ${d.frontmatter.title ?? d.slug}`);
    break;
  }
  case "reindex": {
    reindex(cfg.docsRoot, schema);
    console.log("reindexed");
    break;
  }
  case "validate": {
    const issues = readDocs(dir).flatMap((d) => validateDoc(d, schema));
    issues.forEach((i) => console.error(`${i.file}: ${i.message}`));
    if (issues.length) process.exit(1);
    console.log("valid");
    break;
  }
  case "add": {
    const { title, fields, sections } = parseFlags(rest);
    const file = writeDoc(dir, {
      frontmatter: { title, ...fields },
      title,
      sections,
      order: schema.sections,
    });
    reindex(cfg.docsRoot, schema);
    console.log(`created ${file}`);
    break;
  }
  default:
    console.error(`unknown command: ${cmd}`);
    process.exit(1);
}
```

```tsx
import React, { useState } from "react";
import { Box, Text, useInput, useApp } from "ink";
import TextInput from "ink-text-input";
import type { Doc } from "../core/types.js";
import { searchDocs } from "../core/search.js";

export function App({ docs }: { docs: Doc[] }) {
  const { exit } = useApp();
  const [query, setQuery] = useState("");
  const [cursor, setCursor] = useState(0);
  const results = searchDocs(docs, query);
  const selected = results[Math.min(cursor, results.length - 1)];

  useInput((input, key) => {
    if (key.escape || input === "q") exit();
    else if (key.downArrow) setCursor((c) => Math.min(c + 1, results.length - 1));
    else if (key.upArrow) setCursor((c) => Math.max(c - 1, 0));
  });

  return (
    <Box flexDirection="column">
      <Box>
        <Text>search: </Text>
        <TextInput
          value={query}
          onChange={(v) => {
            setQuery(v);
            setCursor(0);
          }}
        />
      </Box>
      <Box>
        <Box flexDirection="column" width={40} marginRight={2}>
          {results.map((d, i) => (
            <Text key={d.file} inverse={i === cursor}>
              {String(d.frontmatter.title ?? d.slug)}
            </Text>
          ))}
        </Box>
        <Box flexDirection="column" flexGrow={1}>
          {selected && <Text bold>{String(selected.frontmatter.title ?? selected.slug)}</Text>}
          {selected &&
            Object.entries(selected.sections).map(([name, body]) => (
              <Box key={name} flexDirection="column" marginTop={1}>
                <Text underline>{name}</Text>
                <Text>{body}</Text>
              </Box>
            ))}
        </Box>
      </Box>
      <Text dimColor>↑/↓ move · type to search · q/esc quit</Text>
    </Box>
  );
}
```
